charlieh0tel APT repository

APT repository hosted on GitHub Pages. See the source repo for details.

Adding the repository

sudo install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://charlieh0tel.github.io/apt-repo/public.key \
  | sudo gpg --dearmor -o /etc/apt/keyrings/charlieh0tel.gpg
echo "deb [signed-by=/etc/apt/keyrings/charlieh0tel.gpg]" \
  "https://charlieh0tel.github.io/apt-repo bookworm main" \
  | sudo tee /etc/apt/sources.list.d/charlieh0tel.list
sudo apt-get update
